#!/bin/bash
apt-get update -qq || true
DEBIAN_FRONTEND=noninteractive apt-get install -y -qq dotnet-sdk-10.0

# PowerShell for DiGi.Maintenance scripts (Microsoft apt source is blocked)
dotnet tool install -g PowerShell || true
echo 'export PATH="$PATH:$HOME/.dotnet/tools"' >> ~/.bashrc

# Read-only clones of shared repos (select a repo at session start for push access)
for repo in DiGi.Maintenance DiGi.Test; do
  [ -d "/home/user/$repo" ] || \
    GIT_LFS_SKIP_SMUDGE=1 git clone --depth 1 "https://github.com/ZiolkowskiJakub/$repo" "/home/user/$repo"
done
